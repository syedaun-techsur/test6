```js
// A task object:

```

**Implementation:**
```js
const STORAGE_KEY = 'todo-tasks';

/**
 * Load tasks from localStorage.
 * @returns {Array<{id: string, title: string, completed: boolean}>}
 */
export function loadTasks() {
  try {
    const raw = localStorage.getItem(STORAGE_KEY);
    if (!raw) return [];
    const parsed = JSON.parse(raw);
    // Guard: ensure it's an array; ignore corrupted data
    return Array.isArray(parsed) ? parsed : [];
  } catch {
    return [];
  }
}

/**
 * Persist tasks to localStorage.
 * @param {Array<{id: string, title: string, completed: boolean}>} tasks
 */
export function saveTasks(tasks) {
  localStorage.setItem(STORAGE_KEY, JSON.stringify(tasks));
}
```

Key requirements:
- Use ES module `export` (not CommonJS `module.exports`) — `index.html` loads `app.js` as `type="module"`
- `loadTasks()` MUST return `[]` on empty/missing/corrupt data — never `null` or `undefined`
- Storage key must be `'todo-tasks'` exactly — Phase 2 uses the same key
- Try/catch around JSON.parse guards against manually corrupted localStorage

**renderer.js** — Renders a task array to the DOM:

```js
/**
 * Render tasks into #task-list. Toggle #empty-state visibility.
 * @param {Array<{id: string, title: string, completed: boolean}>} tasks
 */
export function renderTasks(tasks) {
  const list = document.getElementById('task-list');
  const emptyState = document.getElementById('empty-state');

  // Clear existing items before re-render
  list.innerHTML = '';

  if (tasks.length === 0) {
    emptyState.hidden = false;
    return;
  }

  emptyState.hidden = true;

  tasks.forEach(task => {
    const li = document.createElement('li');
    li.className = 'task-item';
    li.dataset.id = task.id;

    // Checkbox
    const checkbox = document.createElement('input');
    checkbox.type = 'checkbox';
    checkbox.checked = task.completed;
    checkbox.setAttribute('aria-label', `Mark "${task.title}" complete`);
    // NOTE: onclick handler wired in Phase 2

    // Title
    const title = document.createElement('span');
    title.className = task.completed ? 'task-title completed' : 'task-title';
    title.textContent = task.title;

    // Delete button (styled but not functional until Phase 2)
    const deleteBtn = document.createElement('button');
    deleteBtn.className = 'delete-btn';
    deleteBtn.textContent = '✕';
    deleteBtn.setAttribute('aria-label', `Delete "${task.title}"`);
    // NOTE: onclick handler wired in Phase 2

    li.appendChild(checkbox);
    li.appendChild(title);
    li.appendChild(deleteBtn);
    list.appendChild(li);
  });
}
```

Key requirements for renderer.js:
- Always clear `list.innerHTML = ''` before rendering — prevents duplicates on re-render
- Toggle `emptyState.hidden` — do NOT use CSS display manipulation; the `hidden` HTML attribute is set in Plan 01
- Apply `'task-title completed'` class when `task.completed === true` — this triggers the strikethrough CSS from `styles.css`
- Each `<li>` gets `data-id` attribute — Phase 2 uses this for mutation identification
- Checkbox and delete button are rendered but their event handlers are NOT wired — comment notes `// Phase 2`

**app.js** — Entry point, wires storage + renderer on load:

```js
import { loadTasks } from './storage.js';
import { renderTasks } from './renderer.js';

// Load and render tasks when DOM is ready
document.addEventListener('DOMContentLoaded', () => {
  const tasks = loadTasks();
  renderTasks(tasks);
});
```

Key requirements for app.js:
- Import paths use `./storage.js` and `./renderer.js` (relative, with `.js` extension — required for native ES modules)
- `DOMContentLoaded` listener ensures DOM is ready before querying element IDs
- Phase 2 will add event listeners for the add button and task interactions in this file
